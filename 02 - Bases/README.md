#Ejecutar un módulo sin nodemon:
Ir por terminal a la carpeta donde está el archivo que contiene el módulo 
npm start, o
node src/app.js


#instalar nodemon global
npm install nodemon

#instalar nodemon en dev 
npm install --save-dev nodemon
o
npm install -D nodemon


#Correr nodemon en dev
npm run dev (ver que nodemon esté en la parte dev de packages.json)
